sign it, verify it, break it

a public key checks it, one byte breaks it

$openssl genpkey -algorithm ed25519 -out key.pem
  # (silent) mint the private key
$openssl pkey -in key.pem -pubout -out pub.pem
  # (silent) extract the public half
$printf 'send 100 to bob' > msg.txt  # the message
$openssl pkeyutl -sign -inkey key.pem -rawin \
 -in msg.txt -out sig.bin  # 64-byte signature → sig.bin
$openssl pkeyutl -verify -pubin -inkey pub.pem \
 -rawin -in msg.txt -sigfile sig.bin
✓ Signature Verified Successfully ✓ public key only, no secret
$printf 'send 900 to bob' > msg.txt  # tamper: 1 → 9, one byte
$openssl pkeyutl -verify -pubin -inkey pub.pem \
 -rawin -in msg.txt -sigfile sig.bin
✗ Signature Verification Failure ✗ one byte changed
